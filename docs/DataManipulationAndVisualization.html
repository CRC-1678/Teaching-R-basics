  1. Finally, we might want to check if expression of 2 genes is correlated.
##Let's create a counts matrix with expression of Cdh1 and Tp53
genes_of_interest <- c("Cdh1", "Rdh11")

# Filter and keep in wide format (samples × genes)
counts_two_genes <- counts %>%
  as.data.frame() %>%
  rownames_to_column("Gene") %>%
  filter(Gene %in% genes_of_interest) %>%
  tibble::column_to_rownames("Gene") %>%  # Optional: keep genes as rownames
  t() %>%                                 # Transpose to have samples as rows
  as.data.frame() %>%
  tibble::rownames_to_column("Sample")       

counts_two_genes_with_meta <- left_join(counts_two_genes, metadata, by = "Sample")
head(counts_two_genes_with_meta)
